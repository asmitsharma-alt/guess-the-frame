      setupCloudTransport() {
        if (this.mqttClient) {
          try { this.mqttClient.end(true); } catch(e) {}
        }
        if (this.broadcastChannel) {
          try { this.broadcastChannel.close(); } catch(e) {}
        }

        const topic = 'gtf_game_v1/' + this.roomCode;

        // 1. Local BroadcastChannel for instant same-browser sync
        try {
          this.broadcastChannel = new BroadcastChannel('gtf_bc_' + this.roomCode);
          this.broadcastChannel.onmessage = (event) => {
            if (event && event.data) this.handleIncomingEvent(event.data);
          };
        } catch(e) {}
